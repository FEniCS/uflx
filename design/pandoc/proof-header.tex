\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[L]{\scriptsize\sffamily\color{black!60}PROOF \proofversion}
\fancyfoot[R]{\scriptsize\sffamily\thepage}
